gcloud projects create lpr-car-plate

gcloud config set project lpr-car-plate

gcloud services enable artifactregistry.googleapis.com

gcloud services enable run.googleapis.com

gcloud artifacts repositories create lpr-repo `
    --repository-format=docker `
    --location=asia-southeast1 `
    --project=lpr-car-plate

gcloud auth configure-docker asia-southeast1-docker.pkg.dev

# Step 1: Build amd64 image และ push ไป Artifact Registry ในทีเดียว
docker buildx build --platform linux/amd64 \
  -t asia-southeast1-docker.pkg.dev/lpr-car-plate/lpr-repo/thai-lpr-api \
  --push .

# Step 2: Deploy image นั้นขึ้น Cloud Run
gcloud run deploy thai-lpr-api \
  --image asia-southeast1-docker.pkg.dev/lpr-car-plate/lpr-repo/thai-lpr-api \
  --region asia-southeast1 \
  --platform managed \
  --allow-unauthenticated \
  --cpu 2 \
  --memory 2Gi \
  --concurrency 8 \
  --timeout 60s \
  --min-instances 0 \
  --max-instances 1 \
  --cpu-throttling \
  --port 8080
